	if (document.readyState === 'complete') {
		console.log('options-script.js');

		const colorSelectorNumberStandard = document.getElementById(
			'colorSelectorNumberStandard'
		);

			'colorSelectorBackgroundStandard'
		);

			'colorSelectorNumberWarning'
		);

			'colorSelectorBackgroundTimeout'
		);

		const exampleNumberStandard = document.getElementById(
			'exampleNumberStandard'
		);
		const exampleNumberWarning = document.getElementById(
			'exampleNumberWarning'
		);
		const exampleNumberTimeout = document.getElementById(
			'exampleNumberTimeout'
		);

		// ------------------------------------------------------------------------------------------
		colorSelectorNumberStandard.addEventListener('change', () => {
			console.log(
				'change colorSelectorNumberStandard:',
				examplecolorSelectorNumberStandard.value
			);
		});

			exampleNumberStandard.style.color = colorSelectorNumberStandard.value;

			console.log(

				colorSelectorNumberWarning.value
			);
		});
		colorSelectorNumberWarning.addEventListener('input', () => {
			exampleNumberWarning.style.color = colorSelectorNumberWarning.value;
		});

			console.log(
				'change colorSelectorNumberTimeout:',
				exampleColorSelectorNumberTimeout.value
			);
		});
		colorSelectorNumberTimeout.addEventListener('input', () => {
			exampleNumberTimeout.style.color = colorSelectorNumberTimeout.value;
		});
		// ===========================================================================================
		colorSelectorBackgroundStandard.addEventListener('change', () => {
			console.log(
				'change colorSelectorNumberStandard:',
				colorSelectorBackgroundStandard.value
			);
		});
		colorSelectorBackgroundStandard.addEventListener('input', () => {
			exampleNumberStandard.style.backgroundColor =
				colorSelectorBackgroundStandard.value;
		});
		// ----------------------------------------------------------------------------------------
		colorSelectorBackgroundWarning.addEventListener('change', () => {
			console.log(

			);
		});

			exampleNumberWarning.style.backgroundColor =
